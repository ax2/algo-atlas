// @gcd
function solve(a, b) {
  a = Math.abs(a);
  b = Math.abs(b);
  while (b !== 0) [a, b] = [b, a % b];
  return a;
}
// @sieve
function solve(n) {
  const prime = Array(n + 1).fill(true);
  prime[0] = prime[1] = false;
  for (let p = 2; p * p <= n; p++)
    if (prime[p]) for (let x = p * p; x <= n; x += p) prime[x] = false;
  const out = [];
  for (let p = 2; p <= n; p++) if (prime[p]) out.push(p);
  return out;
}
// @fast-power
function solve(base, exponent, modulus) {
  let a = BigInt(base),
    b = BigInt(exponent),
    m = BigInt(modulus),
    result = 1n % m;
  a = ((a % m) + m) % m;
  while (b > 0n) {
    if (b % 2n) result = (result * a) % m;
    a = (a * a) % m;
    b /= 2n;
  }
  return Number(result);
}
// @bitmask
function solve(n) {
  const subsets = [];
  for (let mask = 0; mask < 2 ** n; mask++) {
    const subset = [];
    for (let bit = 0; bit < n; bit++) if (mask & (1 << bit)) subset.push(bit);
    subsets.push(subset);
  }
  return subsets;
}
// @convex-hull
function solve(points) {
  const p = [...new Map(points.map((x) => [x.join(","), x])).values()].sort(
    (a, b) => a[0] - b[0] || a[1] - b[1],
  );
  if (p.length <= 1) return p;
  const cross = (a, b, c) =>
    (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]);
  function chain(list) {
    const h = [];
    for (const x of list) {
      while (h.length >= 2 && cross(h.at(-2), h.at(-1), x) <= 0) h.pop();
      h.push(x);
    }
    return h.slice(0, -1);
  }
  return chain(p).concat(chain([...p].reverse()));
}
