# @treap
def solve(keys, priorities, queries):
    def left(n):
        r = n['right']
        n['right'], r['left'] = r['left'], n
        return r
    def right(n):
        l = n['left']
        n['left'], l['right'] = l['right'], n
        return l
    def insert(n, key, priority):
        if not n:
            return {'key': key, 'priority': priority, 'left': None, 'right': None}
        if key < n['key']:
            n['left'] = insert(n['left'], key, priority)
            if n['left']['priority'] < n['priority']:
                n = right(n)
        if key > n['key']:
            n['right'] = insert(n['right'], key, priority)
            if n['right']['priority'] < n['priority']:
                n = left(n)
        return n
    root, out = None, []
    for key, priority in zip(keys, priorities):
        root = insert(root, key, priority)
    for key in queries:
        node = root
        while node and node['key'] != key:
            node = node['left' if key < node['key'] else 'right']
        out.append(node is not None)
    return out
# @splay
def solve(values, queries):
    def left(x):
        y = x['right']
        x['right'], y['left'] = y['left'], x
        return y
    def right(x):
        y = x['left']
        x['left'], y['right'] = y['right'], x
        return y
    def splay(root, key):
        if not root or root['key'] == key:
            return root
        if key < root['key']:
            if not root['left']:
                return root
            if key < root['left']['key']:
                root['left']['left'] = splay(root['left']['left'], key)
                root = right(root)
            elif key > root['left']['key']:
                root['left']['right'] = splay(root['left']['right'], key)
                if root['left']['right']:
                    root['left'] = left(root['left'])
            return right(root) if root['left'] else root
        if not root['right']:
            return root
        if key > root['right']['key']:
            root['right']['right'] = splay(root['right']['right'], key)
            root = left(root)
        elif key < root['right']['key']:
            root['right']['left'] = splay(root['right']['left'], key)
            if root['right']['left']:
                root['right'] = right(root['right'])
        return left(root) if root['right'] else root
    root, out = None, []
    for key in values:
        if not root:
            root = {'key': key, 'left': None, 'right': None}
            continue
        root = splay(root, key)
        if root['key'] == key:
            continue
        if key < root['key']:
            node = {'key': key, 'left': root['left'], 'right': root}
            root['left'] = None
        else:
            node = {'key': key, 'left': root, 'right': root['right']}
            root['right'] = None
        root = node
    for key in queries:
        root = splay(root, key)
        out.append(root['key'] if root else None)
    return out
# @lazy-segment
def solve(values, operations):
    n = len(values)
    sums, tags, out = [0] * (4 * n), [0] * (4 * n), []
    def apply(p, l, r, value):
        sums[p] += (r - l) * value
        tags[p] += value
    def push(p, l, r):
        if tags[p] and r - l > 1:
            m = (l + r) // 2
            apply(p * 2, l, m, tags[p])
            apply(p * 2 + 1, m, r, tags[p])
            tags[p] = 0
    def add(p, l, r, a, b, value):
        if b <= l or r <= a:
            return
        if a <= l and r <= b:
            apply(p, l, r, value)
            return
        push(p, l, r)
        m = (l + r) // 2
        add(p * 2, l, m, a, b, value)
        add(p * 2 + 1, m, r, a, b, value)
        sums[p] = sums[p * 2] + sums[p * 2 + 1]
    def query(p, l, r, a, b):
        if b <= l or r <= a:
            return 0
        if a <= l and r <= b:
            return sums[p]
        push(p, l, r)
        m = (l + r) // 2
        return query(p * 2, l, m, a, b) + query(p * 2 + 1, m, r, a, b)
    for i, value in enumerate(values):
        add(1, 0, n, i, i + 1, value)
    for operation in operations:
        op, l, r = operation[:3]
        if op == 'add':
            add(1, 0, n, l, r, operation[3])
        else:
            out.append(query(1, 0, n, l, r))
    return out
# @persistent-segment
def solve(values, updates, queries):
    def build(l, r):
        if r - l == 1:
            return {'sum': values[l]}
        m = (l + r) // 2
        left, right = build(l, m), build(m, r)
        return {'sum': left['sum'] + right['sum'], 'left': left, 'right': right}
    def assign(node, l, r, index, value):
        if r - l == 1:
            return {'sum': value}
        m = (l + r) // 2
        left = assign(node['left'], l, m, index, value) if index < m else node['left']
        right = assign(node['right'], m, r, index, value) if index >= m else node['right']
        return {'sum': left['sum'] + right['sum'], 'left': left, 'right': right}
    def query(node, l, r, a, b):
        if b <= l or r <= a:
            return 0
        if a <= l and r <= b:
            return node['sum']
        m = (l + r) // 2
        return query(node['left'], l, m, a, b) + query(node['right'], m, r, a, b)
    versions = [build(0, len(values))]
    for base, index, value in updates:
        versions.append(assign(versions[base], 0, len(values), index, value))
    return [query(versions[version], 0, len(values), l, r) for version, l, r in queries]
# @sparse-table
def solve(values, queries):
    table, k = [list(values)], 1
    while 2 ** k <= len(values):
        table.append([min(table[k - 1][i], table[k - 1][i + 2 ** (k - 1)]) for i in range(len(values) - 2 ** k + 1)])
        k += 1
    out = []
    for l, r in queries:
        k = (r - l).bit_length() - 1
        out.append(min(table[k][l], table[k][r - 2 ** k]))
    return out
# @kd-tree
def solve(points, target):
    def build(ids, depth):
        if not ids:
            return None
        axis = depth % 2
        ids.sort(key=lambda i: (points[i][axis], i))
        mid = len(ids) // 2
        return {'id': ids[mid], 'axis': axis, 'left': build(ids[:mid], depth + 1), 'right': build(ids[mid + 1:], depth + 1)}
    best, distance = -1, float('inf')
    def visit(node):
        nonlocal best, distance
        if not node:
            return
        p = points[node['id']]
        d = sum((p[i] - target[i]) ** 2 for i in range(2))
        if d < distance or (d == distance and node['id'] < best):
            best, distance = node['id'], d
        delta = target[node['axis']] - p[node['axis']]
        near, far = ('left', 'right') if delta < 0 else ('right', 'left')
        visit(node[near])
        if delta * delta <= distance:
            visit(node[far])
    visit(build(list(range(len(points))), 0))
    return best
